        """Load schema from disk; raise HandoffSchemaError on unsupported keywords."""

        """Return [] if valid, list of per-field errors otherwise."""

    def _check_pattern(self, value: str, pattern: str, field: str) -> str | None: ...

def build_reports_validation_service(workspace_root: Path) -> ReportsValidationService:
    """Compose ReportsValidationService with stdlib validator.

    Schema is read from .dadaia/agentic/schemas/handoff-v1.schema.json.
    Raises WorkspaceNotInitializedError if workspace is not staged.
    """
    schema_path = workspace_root / ".dadaia" / "agentic" / "schemas" / "handoff-v1.schema.json"
    if not schema_path.exists():
        raise WorkspaceNotInitializedError(
            "Schema not found. Run `dadaia public stage` first."
        )
    validator = StdlibHandoffValidator(schema_path)
    reports_root = workspace_root / ".dadaia" / "reports"
    return ReportsValidationService(validator=validator, reports_root=reports_root)

Body sections

# dadaia-handoff-emitter

## When to invoke

Invoke this skill as the LAST action of any session where you produced a report HTML
at `.dadaia/reports/<context>/<agent>/<timestamp>-<slug>.html`.
Do NOT invoke before the HTML is finalized — the content_hash must reflect the final content.

## Steps

### Step 1 — Compute content_hash

Run the following Bash command to obtain the sha256 of the report file:

```bash
sha256sum "<path-to-report.html>"
```

Extract the first field (hex digest). This is `artifact.content_hash`.

### Step 2 — Assemble the handoff JSON

Construct the sidecar object. Required fields:

- `schema_version`: always `"handoff-v1"`
- `agent`: your agent name (e.g. `"software-engineer"`)
- `context`: the active context name (from `primary_context.json` or workflow input)
- `produced_at`: current UTC timestamp in ISO 8601 format (e.g. `"2026-05-16T23:29:05Z"`)
- `artifact.type`: one of `"report"`, `"spec"`, `"task"`, `"implementation"`
- `artifact.path`: relative path from workspace root to the HTML file
- `artifact.content_hash`: the sha256 hex from Step 1

Optional fields (include when applicable):

- `release_id`: the active release id (e.g. `"agent-comms-v1"`)
- `findings[]`: array of `{severity, description}` objects summarising key findings from
  your report. Use severities: `CRITICAL`, `HIGH`, `MEDIUM`, `LOW`, `INFO`.
- `decisions_required[]`: array of strings — each is a question or decision the next agent
  or operator must resolve before proceeding.
- `next_handoff.target_agent`: the next agent in the pipeline (e.g. `"product-engineer"`)
- `next_handoff.expected_output_path`: where the next agent should write its output

### Step 3 — Write the sidecar

The sidecar filename is the report filename with `.html` replaced by `.handoff.json`:

```
<timestamp>-<slug>.html  →  <timestamp>-<slug>.handoff.json
```

Use the Write tool to create the file at that path. The file must be valid JSON.

## I/O contract

- **Input:** the finalized HTML report path + your session context
- **Output:** a `.handoff.json` file at the same directory as the HTML
- **Side effects:** none — pure write, no CLI invocation

## Example

Report produced: `.dadaia/reports/dadaia-workspace/software-engineer/2026-05-16T232905Z-agent-comms-implementation.html`

Sidecar written: `.dadaia/reports/dadaia-workspace/software-engineer/2026-05-16T232905Z-agent-comms-implementation.handoff.json`

Content: see Section 6 FULL fixture in the SE recommendation report.

## Schema reference

The full schema is at `.dadaia/agentic/schemas/handoff-v1.schema.json` (after `dadaia public stage`).
Validate your output: `dadaia reports validate <path-to-sidecar>`
